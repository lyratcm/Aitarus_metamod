namespace = unfallen_empire

country_event = {
	id = unfallen_empire.1
	is_triggered_only = yes
	hide_window = yes

	immediate = {
        #check a day later to allow host flag to be set
        country_event = {
            id = unfallen_empire.11
            days = 1
        }
	}
}

country_event = {
	id = unfallen_empire.11
	is_triggered_only = yes
	hide_window = yes

    trigger = {
        has_origin = unfallen_empire
    }

	immediate = {
        if = {
            limit = {
                has_country_flag = gm_is_host
            }
            country_event = { id = unfallen_empire.2 }
        }
        else = {
            country_event = {
                id = unfallen_empire.12
            }
        }
	}
}

                        }

                        }

                        }

                                owner = root

                                is_star = no

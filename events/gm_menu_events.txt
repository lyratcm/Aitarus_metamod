namespace = gm_menu

# On Game Start (Get Host)
country_event = {
	id = gm_menu.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = no
		NOT = { has_global_flag = gm_host_picked }
	}

    # will trigger on the first country (the host) set the flag then won't trigger a again
	immediate = {
        set_global_flag = gm_host_picked
        set_country_flag = gm_is_host
        country_event = {
            id = unfallen_empire.11
        }
    }
}

	id = gm_menu.2
	is_triggered_only = yes
	hide_window = no

	immediate = {
	}

    #add a set of options, each one will run effect you want to do in that scope (e.g. planet)
    option = {
        # make sure to edit/add this in gm_menu_localisation_l_english.yml to show what will be added
        name = gm_menu.2a
        # prevents adding the modifier more than once (optional)
        allow = {
            not = { has_modifier = seeded_planet_1_modifier }
        }
        add_modifier = { modifier = seeded_planet_1_modifier }
    }
}
